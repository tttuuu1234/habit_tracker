// ユースケースはResult<T>を返し、エラーを明示的に処理 Future<Result<Habit>> execute(CreateHabitParams params) async { if (params.name.isEmpty) { return Result.failure('Name cannot be empty'); } final habit = await _repository.create(params); return Result.success(habit); } // 呼び出し側での網羅的パターンマッチ final result = await createHabitUseCase.execute(params); result.when( success: (habit) => state = AsyncData(habit), failure: (error) => state = AsyncError(error), );
